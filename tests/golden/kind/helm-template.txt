  name: loja
  labels:

    app.kubernetes.io/version: "1.0"
    app.kubernetes.io/managed-by: Helm
    helm.sh/chart: loja-1.0.0

  replicas: 4
  selector:
    matchLabels:
      app.kubernetes.io/name: loja
      app.kubernetes.io/instance: loja

    metadata:
      labels:
        app.kubernetes.io/name: loja
        app.kubernetes.io/instance: loja
    spec:
      containers:
      - name: loja
        image: "kubelearn/loja:1.0"

        - name: http

          httpGet:

            port: http
